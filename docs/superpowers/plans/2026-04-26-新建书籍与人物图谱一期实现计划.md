# 新建书籍与人物图谱一期 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为当前小说工作台补上“新建书籍”独立创建链路，以及可承接初始化实体数据的“人物图谱”第一期能力。

**Architecture:** 先用一个独立的建书输入模型收敛表单与 API 契约，再把它映射到 `Novel / StoryEntity / EntityRelation` 三层数据写入。图谱能力不单独造一套新数据结构，而是复用现有实体与关系表，通过仓储层新增图谱投影查询，分别服务 `/world` 与 `/graph`。

**Tech Stack:** Next.js App Router, React 18, Prisma, Zod, Vitest

```ts
import { z } from "zod";
import type {
  CharacterSeedInput,
  GraphRelationKind,
  NovelCreationInput,
  RelationSeedInput
} from "@/types/domain";

const characterSeedSchema = z.object({
  name: z.string().trim().min(1, "人物姓名不能为空"),
  role: z.string().trim().optional().default(""),
  summary: z.string().trim().optional().default(""),
  factionName: z.string().trim().optional().default(""),
  locationName: z.string().trim().optional().default("")
});

const relationSeedSchema = z.object({
  sourceName: z.string().trim().min(1),
  targetName: z.string().trim().min(1),
  type: z.enum(["ALLY", "ENEMY", "FAMILY", "MENTOR", "SUBORDINATE", "OTHER", "MEMBER_OF", "ROOTED_IN"]),
  description: z.string().trim().optional().default("")
});

export const novelCreationSchema = z.object({
  title: z.string().trim().min(1, "书名不能为空"),
  category: z.string().trim().min(1, "类型不能为空"),
  subGenre: z.string().trim().min(1, "细分类型不能为空"),
  targetAudience: z.string().trim().min(1, "目标受众不能为空"),
  premise: z.string().trim().min(1, "一句话 premise 不能为空"),
  narrativeView: z.string().trim().min(1, "叙事视角不能为空"),
  storyStructure: z.string().trim().min(1, "故事结构不能为空"),
  plannedChapterCount: z.number().int().positive().optional(),
  targetWordsPerChapter: z.number().int().positive().optional(),
  worldSeed: z.string().trim().optional().default(""),
  styleGoal: z.string().trim().optional().default(""),
  characterSeeds: z.array(characterSeedSchema).max(3).default([]),
  relationSeeds: z.array(relationSeedSchema).default([])
});

function uniqueBy<T>(items: T[], keyFn: (item: T) => string) {
  const seen = new Set<string>();
  return items.filter((item) => {
    const key = keyFn(item);
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

export function buildInitialGraphSeed(input: NovelCreationInput) {
  const characterEntities = input.characterSeeds.map((item) => ({
    type: "CHARACTER" as const,
    name: item.name,
    summary: item.summary || item.role || undefined,
    profile: {
      role: item.role || undefined
    }
  }));

  const factionEntities = uniqueBy(
    input.characterSeeds
      .filter((item) => item.factionName)
      .map((item) => ({
        type: "FACTION" as const,
        name: item.factionName!.trim(),
        summary: undefined,
        profile: {}
      })),
    (item) => `${item.type}:${item.name}`
  );

  const locationEntities = uniqueBy(
    input.characterSeeds
      .filter((item) => item.locationName)
      .map((item) => ({
        type: "LOCATION" as const,
        name: item.locationName!.trim(),
        summary: undefined,
        profile: {}
      })),
    (item) => `${item.type}:${item.name}`
  );

  const relationSeeds = [
    ...input.relationSeeds,
    ...input.characterSeeds.flatMap((item): RelationSeedInput[] => {
      const relations: RelationSeedInput[] = [];
      if (item.factionName) {
        relations.push({
          sourceName: item.name,
          targetName: item.factionName,
          type: "MEMBER_OF",
          description: `${item.name} 隶属于 ${item.factionName}`
        });
      }
      if (item.locationName) {
        relations.push({
          sourceName: item.name,
          targetName: item.locationName,
          type: "ROOTED_IN",
          description: `${item.name} 与 ${item.locationName} 有核心联系`
        });
      }
      return relations;
    })
  ];

  return {
    entities: [...characterEntities, ...factionEntities, ...locationEntities],
    relations: relationSeeds
  };
}

export function buildNovelCreationInput(raw: unknown) {
  const input = novelCreationSchema.parse(raw);
  const graph = buildInitialGraphSeed(input);

  return {
    novel: {
      title: input.title,
      premise: input.premise,
      summary: input.premise,
      category: input.category,
      subGenre: input.subGenre,
      targetAudience: input.targetAudience,
      narrativeView: input.narrativeView,
      storyStructure: input.storyStructure,
      plannedChapterCount: input.plannedChapterCount,
      targetWordsPerChapter: input.targetWordsPerChapter,
      worldSeed: input.worldSeed || undefined,
      styleGoal: input.styleGoal || undefined
    },
    graph
  };
}
```

```ts
import { prisma } from "@/lib/prisma";
import { buildNovelCreationInput } from "@/lib/novel-creation";
import type { NovelCreationInput, NovelGraphData } from "@/types/domain";

function toSlug(input: string) {
  return (
    input
      .trim()
      .toLowerCase()
      .replace(/[^a-z0-9\u4e00-\u9fa5]+/g, "-")
      .replace(/^-+|-+$/g, "") || `novel-${Date.now()}`
  );
}

export async function createNovelWithSeedData(raw: unknown) {
  const { novel, graph } = buildNovelCreationInput(raw);
  const slug = toSlug(novel.title);

  const created = await prisma.$transaction(async (tx) => {
    const createdNovel = await tx.novel.create({
      data: {
        slug,
        title: novel.title,
        premise: novel.premise,
        summary: novel.summary,
        category: novel.category,
        subGenre: novel.subGenre,
        targetAudience: novel.targetAudience,
        narrativeView: novel.narrativeView,
        storyStructure: novel.storyStructure,
        plannedChapterCount: novel.plannedChapterCount,
        targetWordsPerChapter: novel.targetWordsPerChapter,
        styleGoal: novel.styleGoal,
        worldSeed: novel.worldSeed,
        genre: novel.category,
        tone: novel.styleGoal
      }
    });

    const entityMap = new Map<string, string>();

    for (const entity of graph.entities) {
      const createdEntity = await tx.storyEntity.create({
        data: {
          novelId: createdNovel.id,
          type: entity.type,
          name: entity.name,
          aliases: [],
          summary: entity.summary,
          tags: [],
          profile: entity.profile
        }
      });

      entityMap.set(entity.name, createdEntity.id);
    }

    for (const relation of graph.relations) {
      const sourceId = entityMap.get(relation.sourceName);
      const targetId = entityMap.get(relation.targetName);

      if (!sourceId || !targetId) {
        continue;
      }

      await tx.entityRelation.create({
        data: {
          sourceId,
          targetId,
          type: relation.type === "MEMBER_OF" || relation.type === "ROOTED_IN" ? "OTHER" : relation.type,
          description: relation.description || undefined
        }
      });
    }

    return createdNovel;
  });

  return created;
}
```

```ts
import { NextResponse } from "next/server";
import { ZodError } from "zod";
import { createNovelWithSeedData, getNovelSummaries } from "@/lib/repositories/novels";

export async function GET() {
  const novels = await getNovelSummaries();
  return NextResponse.json({ novels });
}

export async function POST(request: Request) {
  try {
    const payload = await request.json();
    const novel = await createNovelWithSeedData(payload);
    return NextResponse.json({ novel }, { status: 201 });
  } catch (error) {
    if (error instanceof ZodError) {
      return NextResponse.json(
        {
          error: "建书信息校验失败",
          issues: error.issues
        },
        { status: 400 }
      );
    }

    return NextResponse.json(
      {
        error: "创建书籍失败"
      },
      { status: 500 }
    );
  }
}
```

```ts
export async function getNovelGraphData(novelSlug: string): Promise<NovelGraphData | null> {
  if (!isDatabaseConfigured) {
    return getDemoGraphData(novelSlug);
  }

  const novel = await prisma.novel.findUnique({
    where: { slug: novelSlug },
    include: {
      entities: {
        include: {
          sourceRelations: true,
          targetRelations: true
        }
      },
      chapters: {
        select: { wordCount: true }
      }
    }
  });

  if (!novel) {
    return getDemoGraphData(novelSlug);
  }

  const edges = novel.entities.flatMap((entity) =>
    entity.sourceRelations.map((relation) => ({
      id: relation.id,
      sourceId: relation.sourceId,
      targetId: relation.targetId,
      type: relation.type,
      description: relation.description ?? undefined
    }))
  );

  return {
    novel: {
      id: novel.id,
      slug: novel.slug,
      title: novel.title,
      premise: novel.premise ?? undefined,
      summary: novel.summary ?? undefined,
      genre: novel.genre ?? undefined,
      tone: novel.tone ?? undefined,
      status: novel.status,
      updatedAt: novel.updatedAt.toISOString(),
      chapterCount: novel.chapters.length,
      wordCount: novel.chapters.reduce((sum, chapter) => sum + chapter.wordCount, 0)
    },
    nodes: novel.entities
      .filter((entity) => entity.type === "CHARACTER" || entity.type === "FACTION" || entity.type === "LOCATION")
      .map((entity) => ({
        id: entity.id,
        name: entity.name,
        type: entity.type,
        summary: entity.summary ?? undefined,
        tags: entity.tags
      })),
    edges
  };
}
```
