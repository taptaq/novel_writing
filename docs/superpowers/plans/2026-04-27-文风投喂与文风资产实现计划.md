# 文风投喂与文风资产 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为每本书增加可持续维护的文风资产与文风投喂能力，并让 `续写`、`去 AI 味改写` 默认读取该书的风格记忆。

**Architecture:** 先在 Prisma 和领域类型里补一层 `NovelStyleProfile + NovelStyleSample`，把“原始样文”和“当前生效的文风规则”分开存。然后沿用现有 `novels` 仓储 + App Router API + React 表单模式，补创建链路、独立文风页和 AI prompt 接入，保证无文风资产时仍能平滑退回当前逻辑。

**Tech Stack:** Next.js App Router, React 18, Prisma, Zod, Vitest

- [ ] **Step 4: 实现文风 schema、规范化与请求级生效上下文工具**

在 `src/lib/novel-style.ts` 写入：

```ts
import { z } from "zod";
import type { NovelStyleProfileSummary } from "@/types/domain";

function cleanLines(items: unknown): string[] {
  return Array.isArray(items)
    ? items
        .filter((item): item is string => typeof item === "string")
        .map((item) => item.trim())
        .filter(Boolean)
    : [];
}

export const novelStyleSampleSchema = z.object({
  title: z.string().trim().optional(),
  content: z.string().trim().min(20, "参考样文至少需要 20 个字符"),
  note: z.string().trim().optional()
});

export const novelStyleProfileSchema = z.object({
  styleSummary: z.string().trim().optional().default(""),
  styleRules: z.array(z.string()).default([]),
  avoidRules: z.array(z.string()).default([]),
  dialogueRules: z.array(z.string()).default([]),
  narrationRules: z.array(z.string()).default([]),
  rhythmRules: z.array(z.string()).default([]),
  imageryRules: z.array(z.string()).default([]),
  status: z.enum(["EMPTY", "READY", "FAILED"]).default("EMPTY")
});

export function normalizeStyleProfile(input: unknown): NovelStyleProfileSummary {
  const parsed = novelStyleProfileSchema.parse(input ?? {});

  return {
    styleSummary: parsed.styleSummary || undefined,
    styleRules: cleanLines(parsed.styleRules),
    avoidRules: cleanLines(parsed.avoidRules),
    dialogueRules: cleanLines(parsed.dialogueRules),
    narrationRules: cleanLines(parsed.narrationRules),
    rhythmRules: cleanLines(parsed.rhythmRules),
    imageryRules: cleanLines(parsed.imageryRules),
    status: parsed.status
  };
}

export function buildEffectiveStyleContext(
  profile: NovelStyleProfileSummary | undefined,
  disabled: boolean
) {
  if (disabled || !profile || profile.status !== "READY") {
    return { enabled: false, profile: undefined };
  }

  return { enabled: true, profile };
}
```

import { NextResponse } from "next/server";
import { getNovelStyleWorkspace, updateNovelStyleProfile } from "@/lib/repositories/novels";

export async function GET(_: Request, { params }: { params: { novelSlug: string } }) {
  const workspace = await getNovelStyleWorkspace(params.novelSlug);
  return workspace
    ? NextResponse.json(workspace)
    : NextResponse.json({ message: "Novel not found." }, { status: 404 });
}

export async function PATCH(request: Request, { params }: { params: { novelSlug: string } }) {
  const profile = await updateNovelStyleProfile(params.novelSlug, await request.json());
  return NextResponse.json({ profile });
}

```ts
function buildStyleProfileBlock(input: WritingAssistInput) {
  if (
    input.disableStyleProfile ||
    !input.novel.styleProfile ||
    (input.mode !== "continue" && input.mode !== "rewrite")
  ) {
    return [];
  }

  const profile = input.novel.styleProfile;

  return [
    "",
    "当前作品文风资产：",
    profile.styleSummary ? `- 文风概述：${profile.styleSummary}` : null,
    ...profile.styleRules.map((rule) => `- 正向规则：${rule}`),
    ...profile.avoidRules.map((rule) => `- 避免表达：${rule}`),
    ...profile.dialogueRules.map((rule) => `- 对白规则：${rule}`),
    ...profile.narrationRules.map((rule) => `- 叙述规则：${rule}`)
  ].filter(Boolean) as string[];
}
```
