# 新建书籍设定说明 AI 解析回填 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在新建书籍页支持粘贴文本或上传设定文件，经 AI 解析后先展示结构化预览，再由用户确认回填到表单。

**Architecture:** 方案分成三层：前端新增“设定说明解析区 + 预览回填区”，服务端新增“设定解析接口 + 文件转纯文本工具”，AI 层新增“设定说明结构化抽取器”。创建作品主提交流程保持不变，解析能力只负责生成草稿，不直接创建作品。

**Tech Stack:** Next.js 14、React 18、TypeScript、Zod、现有 AI provider/router、Vitest

```ts
import { z } from "zod";
import { novelLengthOptions } from "@/lib/novel-length";

const setupLengthCategories = novelLengthOptions.map((option) => option.value) as [
  "SHORT",
  "MEDIUM",
  "LONG"
];

export const parsedSetupDraftSchema = z.object({
  title: z.string().trim().optional().default(""),
  category: z.string().trim().optional().default(""),
  subGenre: z.string().trim().optional().default(""),
  targetAudience: z.string().trim().optional().default(""),
  premise: z.string().trim().optional().default(""),
  narrativeView: z.string().trim().optional().default(""),
  storyStructure: z.string().trim().optional().default(""),
  lengthCategory: z.enum(setupLengthCategories).optional().default("MEDIUM"),
  plannedChapterCount: z.number().int().positive().optional(),
  targetWordsPerChapter: z.number().int().positive().optional(),
  worldSeed: z.string().trim().optional().default(""),
  styleGoal: z.string().trim().optional().default(""),
  styleSamples: z.array(z.object({
    title: z.string().trim().optional().default(""),
    content: z.string().trim().min(20),
    note: z.string().trim().optional().default("")
  })).max(3).default([]),
  characterSeeds: z.array(z.object({
    name: z.string().trim().min(1),
    role: z.string().trim().optional().default(""),
    summary: z.string().trim().optional().default(""),
    factionName: z.string().trim().optional().default(""),
    locationName: z.string().trim().optional().default("")
  })).max(3).default([]),
  guessedFields: z.array(z.string()).default([]),
  missingFields: z.array(z.string()).default([]),
  confidenceNotes: z.array(z.string()).default([])
});

export function normalizeParsedSetupDraft(input: unknown) {
  const raw = typeof input === "object" && input !== null ? input : {};
  const normalized = {
    ...raw,
    lengthCategory: raw && typeof raw === "object" && "lengthCategory" in raw ? raw.lengthCategory : "MEDIUM",
    styleSamples: Array.isArray((raw as { styleSamples?: unknown[] }).styleSamples)
      ? (raw as { styleSamples: unknown[] }).styleSamples.slice(0, 3)
      : [],
    characterSeeds: Array.isArray((raw as { characterSeeds?: unknown[] }).characterSeeds)
      ? (raw as { characterSeeds: unknown[] }).characterSeeds.slice(0, 3)
      : []
  };

  return parsedSetupDraftSchema.parse(normalized);
}
```

```ts
export type SetupFileType = "text" | "markdown" | "docx" | "pdf";

export function detectSetupFileType(filename: string): SetupFileType | null {
  const lower = filename.toLowerCase();

  if (lower.endsWith(".txt")) return "text";
  if (lower.endsWith(".md")) return "markdown";
  if (lower.endsWith(".docx")) return "docx";
  if (lower.endsWith(".pdf")) return "pdf";

  return null;
}

export function normalizeSetupSourceText(value: string) {
  return value.replace(/\r\n/g, "\n").trim();
}
```

```ts
import { NextResponse } from "next/server";
import { z } from "zod";
import { buildSetupParsingPrompt } from "@/lib/ai/prompts";
import { normalizeParsedSetupDraft } from "@/lib/novel-setup-parser";

const setupParseRequestSchema = z.object({
  sourceText: z.string().trim().min(20, "设定说明至少需要 20 个字符"),
  sourceName: z.string().trim().optional(),
  sourceType: z.string().trim().optional()
});

export async function POST(request: Request) {
  try {
    const payload = setupParseRequestSchema.parse(await request.json());
    const prompt = buildSetupParsingPrompt(payload.sourceText);

    // 这里后续接现有 AI router，先保持接口结构稳定
    const parsed = normalizeParsedSetupDraft({
      premise: payload.sourceText.slice(0, 60),
      styleSamples: [],
      characterSeeds: [],
      guessedFields: [],
      missingFields: ["title"],
      confidenceNotes: ["当前为最小实现，待接入真实 AI 解析"]
    });

    return NextResponse.json({
      draft: parsed,
      promptPreview: prompt
    });
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json({ error: "设定说明解析参数校验失败", issues: error.issues }, { status: 400 });
    }

    return NextResponse.json({ error: "解析设定说明失败" }, { status: 500 });
  }
}
```

```tsx
function mergeCharacterSeeds(
  incoming: ParsedSetupDraft["characterSeeds"],
  mode: "replace" | "fill-empty"
) {
  if (incoming.length === 0) {
    return form.characterSeeds;
  }

  const nextSeeds =
    mode === "replace"
      ? incoming
      : Array.from({ length: Math.max(form.characterSeeds.length, incoming.length) }).map((_, index) => {
          const current = form.characterSeeds[index] ?? { ...initialCharacterSeed };
          const suggested = incoming[index];

          if (!suggested) {
            return current;
          }

          return {
            name: current.name || suggested.name || "",
            role: current.role || suggested.role || "",
            summary: current.summary || suggested.summary || "",
            factionName: current.factionName || suggested.factionName || "",
            locationName: current.locationName || suggested.locationName || ""
          };
        });

  return nextSeeds.slice(0, 3);
}

function applyParsedSetup(mode: "replace" | "fill-empty") {
  if (!parsedSetupDraft) return;

  setForm((current) => ({
    ...current,
    title: mode === "replace" ? parsedSetupDraft.title ?? "" : current.title || parsedSetupDraft.title || "",
    category: mode === "replace" ? parsedSetupDraft.category ?? "" : current.category || parsedSetupDraft.category || "",
    subGenre: mode === "replace" ? parsedSetupDraft.subGenre ?? "" : current.subGenre || parsedSetupDraft.subGenre || "",
    targetAudience: mode === "replace" ? parsedSetupDraft.targetAudience ?? "" : current.targetAudience || parsedSetupDraft.targetAudience || "",
    premise: mode === "replace" ? parsedSetupDraft.premise ?? "" : current.premise || parsedSetupDraft.premise || "",
    narrativeView: mode === "replace" ? parsedSetupDraft.narrativeView ?? current.narrativeView : current.narrativeView || parsedSetupDraft.narrativeView || current.narrativeView,
    storyStructure: mode === "replace" ? parsedSetupDraft.storyStructure ?? current.storyStructure : current.storyStructure || parsedSetupDraft.storyStructure || current.storyStructure,
    worldSeed: mode === "replace" ? parsedSetupDraft.worldSeed ?? "" : current.worldSeed || parsedSetupDraft.worldSeed || "",
    styleGoal: mode === "replace" ? parsedSetupDraft.styleGoal ?? "" : current.styleGoal || parsedSetupDraft.styleGoal || "",
    characterSeeds: mergeCharacterSeeds(parsedSetupDraft.characterSeeds, mode)
  }));
}
```
