```ts
export async function saveChapterDraft(
  novelSlug: string,
  chapterSlug: string,
  input: { plainText: string; source: "manual" | "autosave"; note?: string }
) {
  const normalizedText = input.plainText.trim();
  const wordCount = estimateWordCount(normalizedText);

  return prisma.$transaction(async (tx) => {
    const chapter = await tx.chapter.findFirst({
      where: {
        slug: chapterSlug,
        novel: {
          slug: novelSlug
        }
      }
    });

    if (!chapter) {
      throw new Error("CHAPTER_NOT_FOUND");
    }

    const updatedChapter = await tx.chapter.update({
      where: { id: chapter.id },
      data: {
        plainText: normalizedText,
        content: normalizedText,
        wordCount
      }
    });

    const version = await tx.chapterVersion.create({
      data: {
        chapterId: chapter.id,
        source: input.source,
        note: input.note ?? null,
        content: normalizedText,
        plainText: normalizedText
      }
    });

    return {
      chapter: {
        slug: updatedChapter.slug,
        wordCount: updatedChapter.wordCount,
        updatedAt: updatedChapter.updatedAt.toISOString()
      },
      version: {
        id: version.id,
        source: version.source,
        createdAt: version.createdAt.toISOString(),
        wordCount
      }
    };
  });
}
```

```ts
const payloadSchema = z.object({
  plainText: z.string(),
  source: z.enum(["manual", "autosave"]),
  note: z.string().trim().optional()
});

export async function PATCH(
  request: Request,
  { params }: { params: { novelSlug: string; chapterSlug: string } }
) {
  try {
    const payload = payloadSchema.parse(await request.json());
    const result = await saveChapterDraft(params.novelSlug, params.chapterSlug, payload);
    return NextResponse.json(result);
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json({ message: "Invalid chapter payload." }, { status: 400 });
    }

    if (error instanceof Error && error.message === "CHAPTER_NOT_FOUND") {
      return NextResponse.json({ message: "Chapter not found." }, { status: 404 });
    }

    return NextResponse.json({ message: "Saving chapter failed." }, { status: 500 });
  }
}
```

```ts
export async function createChapter(novelSlug: string) {
  return prisma.$transaction(async (tx) => {
    const novel = await tx.novel.findUnique({
      where: { slug: novelSlug },
      include: {
        chapters: {
          orderBy: {
            order: "desc"
          },
          take: 1
        }
      }
    });

    if (!novel) {
      throw new Error("NOVEL_NOT_FOUND");
    }

    const nextOrder = (novel.chapters[0]?.order ?? 0) + 1;
    const title = `第 ${nextOrder} 章`;
    const slug = await createUniqueChapterSlug(tx, novel.id, `chapter-${nextOrder}`);

    const chapter = await tx.chapter.create({
      data: {
        novelId: novel.id,
        slug,
        title,
        order: nextOrder,
        content: "",
        plainText: "",
        wordCount: 0
      }
    });

    return {
      chapter: {
        slug: chapter.slug,
        title: chapter.title,
        order: chapter.order
      }
    };
  });
}
```
