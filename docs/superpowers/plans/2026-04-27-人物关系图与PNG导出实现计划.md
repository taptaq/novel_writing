> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Create `src/lib/graph-view.ts`:

```ts
import type { GraphEdgeSummary, GraphNodeSummary, NovelGraphData } from "@/types/domain";

export type GraphScope = "characters" | "characters-factions" | "all";

const scopeKinds: Record<GraphScope, GraphNodeSummary["type"][]> = {
  characters: ["CHARACTER"],
  "characters-factions": ["CHARACTER", "FACTION"],
  all: ["CHARACTER", "FACTION", "LOCATION"]
};

export function getAvailableGraphScopes(graph: NovelGraphData): GraphScope[] {
  const types = new Set(graph.nodes.map((node) => node.type));
  const scopes: GraphScope[] = [];

  if (types.has("CHARACTER")) scopes.push("characters");
  if (types.has("CHARACTER") || types.has("FACTION")) scopes.push("characters-factions");
  if (types.has("CHARACTER") || types.has("FACTION") || types.has("LOCATION")) scopes.push("all");

  return scopes;
}

export function getDefaultGraphScope(graph: NovelGraphData): GraphScope {
  return graph.nodes.some((node) => node.type === "CHARACTER") ? "characters" : "all";
}

export function filterGraphByScope(graph: NovelGraphData, scope: GraphScope): NovelGraphData {
  const visibleKinds = new Set(scopeKinds[scope]);
  const nodes = graph.nodes.filter((node) => visibleKinds.has(node.type));
  const nodeIds = new Set(nodes.map((node) => node.id));
  const edges = graph.edges.filter(
    (edge) => nodeIds.has(edge.sourceId) && nodeIds.has(edge.targetId)
  );

  return { ...graph, nodes, edges };
}

export function buildGraphDetailState(graph: NovelGraphData, selectedNodeId?: string) {
  if (!selectedNodeId) {
    return {
      kind: "overview" as const,
      node: undefined,
      relations: [],
      counts: { nodes: graph.nodes.length, edges: graph.edges.length }
    };
  }

  const node = graph.nodes.find((item) => item.id === selectedNodeId);
  if (!node) {
    return {
      kind: "overview" as const,
      node: undefined,
      relations: [],
      counts: { nodes: graph.nodes.length, edges: graph.edges.length }
    };
  }

  const nodeById = new Map(graph.nodes.map((item) => [item.id, item]));
  const relations = graph.edges
    .filter((edge) => edge.sourceId === selectedNodeId || edge.targetId === selectedNodeId)
    .map((edge) => {
      const targetId = edge.sourceId === selectedNodeId ? edge.targetId : edge.sourceId;
      return {
        edge,
        target: nodeById.get(targetId)!
      };
    });

  return {
    kind: "node" as const,
    node,
    relations,
    counts: { nodes: graph.nodes.length, edges: graph.edges.length }
  };
}
```

```ts
import type { GraphScope } from "@/lib/graph-view";

const scopeLabelMap: Record<GraphScope, string> = {
  characters: "人物",
  "characters-factions": "人物 + 势力",
  all: "人物 + 势力 + 地点"
};

export function buildGraphExportFilename(slug: string, scope: GraphScope, now = new Date()) {
  const date = now.toISOString().slice(0, 10);
  return `${slug}-${scope}-${date}.png`;
}

export function getGraphExportLabel(title: string, scope: GraphScope, now = new Date()) {
  const date = now.toISOString().slice(0, 10);
  return `${title} · ${scopeLabelMap[scope]} · ${date}`;
}
```

```tsx
"use client";

import { useMemo, useState } from "react";
import { buildGraphDetailState, filterGraphByScope, getAvailableGraphScopes, getDefaultGraphScope } from "@/lib/graph-view";
import type { NovelGraphData } from "@/types/domain";
import { GraphDetailPanel } from "@/components/graph-detail-panel";
import { GraphFilterPanel } from "@/components/graph-filter-panel";

export function GraphWorkspace({ graph }: { graph: NovelGraphData }) {
  const [scope] = useState(() => getDefaultGraphScope(graph));
  const [selectedNodeId] = useState<string | undefined>(undefined);
  const scopes = useMemo(() => getAvailableGraphScopes(graph), [graph]);
  const filteredGraph = useMemo(() => filterGraphByScope(graph, scope), [graph, scope]);
  const detail = useMemo(() => buildGraphDetailState(filteredGraph, selectedNodeId), [filteredGraph, selectedNodeId]);

  return (
    <div className="graph-shell">
      <GraphFilterPanel scope={scope} scopes={scopes} />
      <section className="graph-canvas panel">
        <div className="panel-header">
          <div>
            <p className="panel-eyebrow">关系画布</p>
            <h2>{graph.novel.title}</h2>
          </div>
        </div>
        <div className="graph-canvas-stage">关系图画布</div>
      </section>
      <GraphDetailPanel detail={detail} />
    </div>
  );
}
```

Create `src/components/graph-canvas.tsx`:

```tsx
"use client";

import { useEffect, useRef } from "react";
import cytoscape, { type Core } from "cytoscape";
import type { NovelGraphData } from "@/types/domain";

export function GraphCanvas({
  graph,
  selectedNodeId,
  onSelectNode,
  onReady
}: {
  graph: NovelGraphData;
  selectedNodeId?: string;
  onSelectNode: (nodeId?: string) => void;
  onReady?: (core: Core) => void;
}) {
  const hostRef = useRef<HTMLDivElement | null>(null);
  const coreRef = useRef<Core | null>(null);

  useEffect(() => {
    if (!hostRef.current) return;

    const core = cytoscape({
      container: hostRef.current,
      elements: [
        ...graph.nodes.map((node) => ({
          data: {
            id: node.id,
            label: node.name,
            type: node.type
          }
        })),
        ...graph.edges.map((edge) => ({
          data: {
            id: edge.id,
            source: edge.sourceId,
            target: edge.targetId,
            type: edge.type
          }
        }))
      ],
      layout: {
        name: "cose",
        animate: false,
        fit: true
      },
      style: [
        {
          selector: "node",
          style: {
            label: "data(label)",
            "background-color": "#d9c3a5",
            color: "#3b2b1b",
            "text-valign": "center",
            "text-halign": "center",
            "font-size": 12,
            width: 52,
            height: 52
          }
        },
        {
          selector: 'node[type = "CHARACTER"]',
          style: { "background-color": "#20324a", color: "#ffffff" }
        },
        {
          selector: "edge",
          style: {
            width: 2,
            "line-color": "#ccb192",
            "target-arrow-color": "#ccb192",
            "target-arrow-shape": "triangle"
          }
        },
        {
          selector: ".dimmed",
          style: {
            opacity: 0.2
          }
        },
        {
          selector: ".focused",
          style: {
            opacity: 1,
            "line-color": "#c9782a",
            "target-arrow-color": "#c9782a"
          }
        }
      ]
    });

    core.on("tap", "node", (event) => {
      onSelectNode(event.target.id());
    });

    core.on("tap", (event) => {
      if (event.target === core) {
        onSelectNode(undefined);
      }
    });

    coreRef.current = core;
    onReady?.(core);

    return () => {
      core.destroy();
      coreRef.current = null;
    };
  }, [graph, onReady, onSelectNode]);

  useEffect(() => {
    const core = coreRef.current;
    if (!core) return;

    core.elements().removeClass("dimmed focused");
    if (!selectedNodeId) return;

    const selected = core.getElementById(selectedNodeId);
    const neighborhood = selected.closedNeighborhood();
    core.elements().difference(neighborhood).addClass("dimmed");
    neighborhood.addClass("focused");
  }, [selectedNodeId]);

  return <div ref={hostRef} className="graph-canvas-stage" />;
}
```
